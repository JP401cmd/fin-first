Option Explicit

Sub BepaalFIRE()
    Dim wsP As Worksheet, wsES As Worksheet
    Set wsP = ThisWorkbook.Worksheets("P")
    Set wsES = ThisWorkbook.Worksheets("ES")
    Dim leeftijd As Double
    leeftijd = wsP.Range("B7").Value

    Dim prevCalc As XlCalculation
    prevCalc = Application.Calculation
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    On Error GoTo Herstel

    ' Pensioenleeftijd: FIRE = AOW-leeftijd, geen insluiting
    If wsES.Range("C7").Value = "pensioen" Then
        wsP.Range("B16").Value = wsES.Range("C15").Value
        Application.Calculate
        GoTo ToonStatus
    End If

    ' Opeten / Nalaten / Eeuwigdurend: bisectie op de gap (P!B38 -> 0)
    Dim loM As Long, hiM As Long, midM As Long
    loM = 0
    hiM = CLng((100 - leeftijd) * 12)
    wsP.Range("B16").Value = leeftijd + hiM / 12
    Application.Calculate
    If wsP.Range("B38").Value < 0 Then
        ' Niet haalbaar: B16 blijft op de horizon staan -> B38 < 0
        ' -> B93 = unreachable_within_horizon, B94 bevat de extra-sparen-hint (B96)
        GoTo ToonStatus
    End If
    Do While hiM - loM > 1
        midM = (loM + hiM) \ 2
        wsP.Range("B16").Value = leeftijd + midM / 12
        Application.Calculate
        If wsP.Range("B38").Value >= 0 Then hiM = midM Else loM = midM
    Loop
    wsP.Range("B16").Value = leeftijd + hiM / 12
    Application.Calculate

ToonStatus:
    ' === Statuscellen uitsluitend NA de bisectie uitlezen ===
    Dim solverStatus As String, melding As String
    solverStatus = CStr(wsP.Range("B93").Value)
    melding = CStr(wsP.Range("B94").Value)
    wsP.Range("B100").Value = solverStatus   ' laatste solver-status

    Application.Calculation = prevCalc
    Application.ScreenUpdating = True

    If solverStatus = "unreachable_within_horizon" _
       Or solverStatus = "pension_shortfall" Then
        MsgBox melding, vbExclamation, "FIRE-solver"
    Else
        MsgBox melding, vbInformation, "FIRE-solver"
    End If
    Exit Sub

Herstel:
    Application.Calculation = prevCalc
    Application.ScreenUpdating = True
    MsgBox "BepaalFIRE is gestopt met een fout: " & Err.Description, _
           vbCritical, "FIRE-solver"
End Sub

Sub RunScenarioBand()
    Dim wsP As Worksheet, wsSim As Worksheet
    Set wsP = ThisWorkbook.Worksheets("P")
    Set wsSim = ThisWorkbook.Worksheets("Sim")
    Dim sc As Variant: sc = Array("Pessimistisch", "Verwacht", "Optimistisch")
    Dim leeftijd As Double: leeftijd = wsP.Range("B7").Value
    Dim maxM As Long: maxM = CLng((100 - leeftijd) * 12)
    Dim i As Long, loM As Long, hiM As Long, midM As Long
    For i = 0 To 2
        wsP.Range("B42").Value = sc(i)
        wsP.Range("B16").Value = leeftijd + maxM / 12
        Application.Calculate
        If wsP.Range("B38").Value < 0 Then
            wsSim.Range("B6").Offset(i, 0).Value = CVErr(xlErrNA)  ' niet haalbaar
            wsSim.Range("C6").Offset(i, 0).Value = CVErr(xlErrNA)
            wsSim.Range("D6").Offset(i, 0).Value = CVErr(xlErrNA)
        Else
            loM = 0: hiM = maxM
            Do While hiM - loM > 1
                midM = (loM + hiM) \ 2
                wsP.Range("B16").Value = leeftijd + midM / 12
                Application.Calculate
                If wsP.Range("B38").Value >= 0 Then hiM = midM Else loM = midM
            Loop
            wsP.Range("B16").Value = leeftijd + hiM / 12
            Application.Calculate
            wsSim.Range("B6").Offset(i, 0).Value = wsP.Range("B16").Value
            wsSim.Range("C6").Offset(i, 0).Value = ThisWorkbook.Worksheets("Rapport").Range("L6").Value
            wsSim.Range("D6").Offset(i, 0).Value = wsSim.Range("B3").Value
        End If
    Next i
    ' herstel basisstand: Verwacht + bijbehorende FIRE-leeftijd
    wsP.Range("B42").Value = "Verwacht"
    If IsNumeric(wsSim.Range("B7").Value) Then wsP.Range("B16").Value = wsSim.Range("B7").Value
    Application.Calculate
End Sub

Sub RunMonteCarlo()
    Dim n As Long, i As Long
    n = Sheets("MC").Range("B1").Value
    Sheets("MC").Range("B5").Value = 1
    Sheets("MC").Range("B14:B1013").ClearContents
    For i = 1 To n
        Sheets("MC").Range("B11").Value = i
        Application.Calculate
        Sheets("MC").Cells(13 + i, 2).Value = Sheets("MC").Range("B8").Value
    Next i
    Sheets("MC").Range("B5").Value = 0
    Sheets("MC").Range("B11").Value = 1
End Sub
